interface DetectionLog {
  id: string; time: string; sender: string; subject: string;
  riskScore: number; riskLevel: "high" | "medium" | "low" | "none";
  spoofTypes: ("brand_keyword_abuse"|"domain_typosquatting"|"newly_registered_domain"|"display_name_spoof")[];
  impersonatedEntity: { type: "brand" | "displayname"; name: string } | null;
  action: "isolated" | "blocked" | "dropped" | "audit" | "passed";
}
// 保护对象（display-name-spoof-detection.tsx · ProtectedObject）
interface ProtectedObject {
  id; name; email; title;
  category: "executive"|"finance"|"business"|"hr"|"tech"|"custom";
  level: "high"|"medium"|"low"; observe: boolean;
  baseline: "not_started"|"learning"|"learned"|"failed";
  baselineDay; baselineSamples; todayHit; todayQuarantine; todayTag;
  thresholds:{observe;standard;strict}; customThreshold: boolean;
  rules: { id; displayName; email; matchType:"exact"|"wildcard"|"regex"; bonus }[];
  sensitivity?; confidenceThreshold?; disposalMode?; disposalAction?; markStyle?; notify?; adminEmails?;
}
