4. 数据模型与 API 映射

4.1 前端数据模型（依 demo state 归纳）

interface RecipientCheckConfig {
  enabled: boolean                       // 模块开关（recipientCheckEnabled，默认 true）
  limit: {
    enabled: boolean                     // 数量限制策略开关（默认 true）
    mode: "detailed" | "merged"          // 配置模式（默认 detailed）
    inbound:  { max: number; scope: "local" | "all"; action: Action }   // 30 / local / block
    outbound: { max: number; action: Action }                            // 50 / review（detailed 模式）
    internal: { max: number; action: Action }                            // 20 / quarantine（detailed 模式）
    merged:   { max: number; action: Action }                            // 50 / review（merged 模式，外发+域内共用）
  }
  existence: {
    enabled: boolean                     // 存在性验证开关（默认 true）
    action: Action                       // 验证失败动作（默认 block）；仅接收方向生效
  }
}
type Action = "quarantine" | "review" | "block" | "drop"  // 阈值范围 1-1000（demo input min/max）
