url_scan_logs：scan_id(PK) / mail_id(FK) / url / domain / url_type(phishing|malware|c2|spam|qr_phishing)
              / url_verdict(safe|suspicious|malicious) / reputation_score / scan_time / block_action
attachment_logs：scan_id(PK) / mail_id(FK) / file_name / file_type / file_hash_sha256 / av_result(clean|virus)
              / sandbox_result(clean|suspicious|malicious|detecting) / threat_type(virus|macro|zip_bomb|exploit)
              / detection_engine / scan_time / initial_action
domain_blacklist：domain(PK) / add_time / add_by / source(manual|auto) / reason
sample_library：file_hash_sha256(PK) / file_name / file_type / first_seen / detection_count / sandbox_report_url / threat_family
