type IntentType = "porn_gambling" | "political" | "phishing" | "spam" | "subscription"
type DetectionMode = "classification" | "threshold"

interface ThresholdSegment { id: string; min: number; max: number; action: ActionType }
interface MarkDeliverConfig {
  deliveryTarget: "inbox" | "spam_folder"
  subjectMark: { enabled: boolean; text: string; position: "prefix" | "suffix" }
  bodyMark:    { enabled: boolean; text: string; position: "prefix" | "suffix" }
}
interface IntentConfig {
  enabled: boolean
  mode: DetectionMode
  classificationAction: ActionType
  thresholdSegments: ThresholdSegment[]
  markDeliverConfig: MarkDeliverConfig
}
type DirectionIntentConfig = Record<IntentType, IntentConfig>
interface IntentEngineConfig { receive: DirectionIntentConfig; send: DirectionIntentConfig; internal: DirectionIntentConfig }
