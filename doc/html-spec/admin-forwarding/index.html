模块定位网关级邮件路由集中管控：入站投递（收信域映射）、内部转发（Relay 规则）、出站投递（代理 IP → 投递通道 → 路由规则三层模型 + TLS 策略）、发信身份认证（SMTP/LDAP/POP3/IMAP 按域名作用域）四类能力
能力位platform，scope=platform，租户管理员任何形态均不可见（tenantAccess=hidden）
网关现状⚠ 部分能力有底层对应但无 UI 化 API：收信域≈postfix relay_domains/transport 查询通道；发信认证≈authd（local/smtp_relay/ldap 后端）；出站路由≈osg-proxysend（proxysvr 出站模块）。四 Tab 的 CRUD/探测 API 与六类配置表均不存在，且存在多处底层冲突（authd 明文 fail-closed、优先级语义相反），详见 §4 / §10

type ProbeStatus = "normal" | "abnormal" | "unchecked" | "partial"
type EnableStatus = "enabled" | "disabled"

interface DomainMapping {            // 收信域（F1）
  id: string; domainName: string
  targetHosts: string[]              // 多目的地址，轮询负载（暂不支持权重）
  targetPort: number
  status: ProbeStatus; abnormalCount?: number   // partial 时异常 IP 数
  lastProbeTime: string
}

type RcptMatchType = "contains" | "equals" | "regex"
interface RelayRule {                // 转发规则（F2）
  id: string; ruleName: string; priority: number     // 数值越小越优先
  sourceIp: string                   // ALL / 单IP / CIDR / 多IP逗号分隔
  useSpf: boolean                    // 与来源 IP 为 OR：任一命中即通过来源认证
  heloValue: string; fromDomain: string
  rcptDomain: string; rcptMatchType: RcptMatchType
  spamFilter: boolean; status: EnableStatus; updatedAt: string
}

type TlsVersion = "TLSv1.0" | "TLSv1.1" | "TLSv1.2" | "TLSv1.3"
type CipherProfile = "default" | "high" | "compatible"
interface OutboundProxy {            // 代理 IP（F3）
  id: string; name: string           // 2-50 字符唯一
  proxyIp: string; proxyPort: number // 默认端口 6620
  egressIp: string                   // SMTP 外发源 IP
  heloHostname: string               // 空=系统默认 mail.gateway.local
  tlsMinVersion: TlsVersion; cipherProfile: CipherProfile
  status: EnableStatus; probeStatus: ProbeStatus; updatedAt: string
}

interface OutboundChannel {          // 投递通道（F4）
  id: string; channelName: string; status: EnableStatus
  proxyIds: string[]                 // 有序 = 通道内优先级
}

type TlsLevel = "plain" | "prefer" | "force" | "forceVerify"
interface OutboundRule {             // 出站路由规则（F5）
  id: string; ruleName: string; priority: number
  sourceIp: string; fromDomain: string; fromUser: string
  rcptDomain: string; rcptDomainMatch: RcptMatchType
  rcptUser: string; rcptUserMatch: "to" | "cc" | "bcc"
  targetHost: string; targetPort: number   // 空=按收信域 MX 投递
  tlsLevel: TlsLevel
  channelId: string                  // "default" 或通道 id
  status: EnableStatus
  tlsSuccessRate: number             // 最近 24h TLS 握手成功率(%)
}

type AuthProtocol = "SMTP" | "LDAP" | "POP3" | "IMAP"
type AuthScene = "userSpace" | "smtpSend" | "mailSync"
type AuthTlsMode = "off" | "prefer" | "force"
interface AuthConfig {               // 发信认证（F6）
  id: string; priority: number       // 数字越小优先级越高
  domainScope: string[]              // ["All"] 或具体域名（互斥）
  authProtocol: AuthProtocol; serverHost: string; serverPort: number
  tlsMode: AuthTlsMode               // off=明文；prefer=可降级；force=必须TLS
  verifyCert: boolean                // tlsMode=off 时无意义（UI 置灰）
  authTimeout: number                // 1-300 秒
  scenes: AuthScene[]; updatedAt: string
}

5. 业务逻辑规格（以 PRD 为准，demo 行为已实测标注）

5.1 匹配与数据流转（PRD §1.3/§6.1）

  • 入站：外部投递 → 收信域匹配目标主机/端口 → 转发规则按优先级数值升序匹配、命中即止 → 按规则 spamFilter 决定是否过反垃圾 → 投递内部。
  • 出站：本域外发 → 出站路由规则按优先级升序匹配 → 选定投递通道/目标主机 → 按 TLS 等级建连 → 经通道内代理 IP（有序=通道内优先级）出口投递。demo 模拟器展示完整链路：规则 → 通道 → 代理（出口 IP）→ HELO → TLS 策略 → 预计响应 250 2.0.0 Ok。
  • 认证：用户空间认证 / SMTP 发信认证 / 邮件同步代理三场景，按域名作用域 + 优先级（数字小者先）命中；同域名+同场景仅允许一条生效配置（demo 抽屉内实时冲突校验，含 All 交叠判定）。
  • 来源认证 OR 语义（转发规则）：来源 IP 与「使用 SPF 记录」为 OR，任一命中即通过；其余条件 AND。demo 中 SPF 授权简化为发信域名包含匹配（真实实现应解析该域 SPF 记录得出授权 IP 集）。
  • 引用一致性：代理被删除 → 通道自动剔除该代理仅由正常代理负载；通道被删除 → 引用规则显示「通道失效」徽章并 fallback 默认通道（demo 行为；PRD 要求删除拦截，§9-D6/§10-Q7）。
  • 完成度联动：四类配置完成度联动租户中心「路由设置」接入状态（待接流 → 已接入）（PRD §5.3，多租户形态）。
