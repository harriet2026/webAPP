1. 页面布局结构

发信行为管控不是独立路由，而是「身份认证与仿冒检测」策略页（/filter-rules/identity）左侧流水线导航中的一个策略节点。点击「发信行为管控」节点，右侧策略卡内渲染 SenderBehaviorControlModule（showHeader=false）。策略卡的标题「发信行为管控」与「已启用」开关由父页面渲染，非模块自身。

interface BehaviorRule {
  id: string; name: string
  direction: "inbound"|"outbound"|"internal"|"bidirectional"
  objectType: "global"|"sender"|"senderIp"|"senderDomain"
  senderSubType?: "individual"|"group"|"organization"
  senderEmail?; senderGroupId?; senderGroupName?; senderOrgId?; senderOrgName?
  ipSubType?: "single"|"ipGroup"; ipAddress?; ipGroupId?; ipGroupName?; domain?
  timeWindow: "1min"|"5min"|"15min"|"1hour"|"6hour"|"24hour"|"day"
  dimensionA: "ip_count"|"recipient_count"|"mail_count"|"attachment_size"
  thresholdA: number; enableOrCondition: boolean; dimensionB?; thresholdB?
  action: "review"|"quarantine"|"drop"|"block"
  priority: number; expireTime?; remark?; enabled: boolean
  createdAt: string; modifiedAt: string
}
